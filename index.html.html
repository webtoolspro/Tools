// TOOLS DATA

const tools = [

  { cat: "content", title: "Topic Gap Finder", desc: "Find the topics your competitors wrote — you didn't", link: "https://topicgapanalyzer.lovable.app/", icon: "🧩" },

  { cat: "content", title: "Topic Outline Gap Finder", desc: "Build Topic Outline that Ranks #1 on Google", link: "https://topicsoutlinegapfinder.lovable.app/", icon: "📌" },

  { cat: "seo", title: "Meta Tags Generator", desc: "Create meta tags that outrank the competition", link: "https://metatagsgenerator.lovable.app", icon: "🏷️" },

  { cat: "seo", title: "SEO Audit Generator", desc: "Find out why you're not ranking #1 on Google", link: "https://webpageaudittool.lovable.app/", icon: "🔎" },

  { cat: "content", title: "Human Content Writer", desc: "Write blog articles that outrank the top results.", link: "https://toolkitzy.com/human-content-writer/", icon: "✍️" },

  { cat: "analysis", title: "Webpage Indexing Checker", desc: "Check if your pages are indexed on Google", link: "https://webpageindexingchecker.lovable.app/", icon: "🌐" },

  { cat: "seo", title: "AI Schema Generator", desc: "Generate JSON-LD Schema from any webpage URL", link: "https://schemacodegenerator.lovable.app/", icon: "🤖" },

  { cat: "analysis", title: "Topical Authority Builder", desc: "Build a complete topical authority map in seconds", link: "https://topicalauthoritybuilder.lovable.app/", icon: "🏗️" }

];

const extendedTools = [...tools];

let currentPage = 1;

let perPage = 10;

let filteredTools = [...extendedTools];

let activeTag = "all";

const cardsContainer = document.getElementById('cards');

const paginationContainer = document.getElementById('pagination');

const perPageSelect = document.getElementById('perPageSelect');

const searchInput = document.getElementById('search');

function getIconForTool(tool) {

  if (tool.icon) return tool.icon;

  const catIconMap = { content: "📝", seo: "🔍", marketing: "📢", analysis: "📈" };

  return catIconMap[tool.cat] || "🛠️";

}

function escapeHtml(str) {

  if (!str) return '';

  return str.replace(/[&<>]/g, function(m) {

    if (m === '&') return '&amp;';

    if (m === '<') return '&lt;';

    if (m === '>') return '&gt;';

    return m;

  });

}

function renderTools() {

  cardsContainer.innerHTML = '';

  const start = (currentPage - 1) * perPage;

  const end = start + perPage;

  const pageItems = filteredTools.slice(start, end);

  if (pageItems.length === 0) {

    cardsContainer.innerHTML = `<div style="grid-column:1/-1; text-align:center; padding: 60px 20px; background:#FAFAFE; border-radius: 48px;"><i class="fas fa-magic" style="font-size: 42px; color:#CBD5E1;"></i><h3 style="margin-top:16px; font-weight:600;">No tools found</h3><p style="color:#64748B;">Try adjusting your search or category</p></div>`;

    return;

  }

  pageItems.forEach(tool => {

    const card = document.createElement('div');

    card.className = 'card';

    const iconSymbol = getIconForTool(tool);

    const categoryBadge = tool.cat.charAt(0).toUpperCase() + tool.cat.slice(1);

    card.innerHTML = `

      <div class="card-icon">${iconSymbol}</div>

      <h3>${escapeHtml(tool.title)}</h3>

      <p>${escapeHtml(tool.desc)} <span style="font-size: 11px; background:#F1F4F9; padding:2px 8px; border-radius: 40px; color:#475569;">${categoryBadge}</span></p>

      <div class="card-arrow"><i class="fas fa-arrow-right"></i></div>

    `;

    card.addEventListener('click', (e) => {

      e.stopPropagation();

      window.open(tool.link, '_blank');

    });

    cardsContainer.appendChild(card);

  });

}

function renderPagination() {

  paginationContainer.innerHTML = '';

  const totalPages = Math.ceil(filteredTools.length / perPage);

  if (totalPages <= 1) return;

  const maxVisible = 5;

  let startPage = Math.max(1, currentPage - Math.floor(maxVisible / 2));

  let endPage = Math.min(totalPages, startPage + maxVisible - 1);

  if (endPage - startPage + 1 < maxVisible) startPage = Math.max(1, endPage - maxVisible + 1);

  if (startPage > 1) {

    addPageButton(1);

    if (startPage > 2) addEllipsis();

  }

  for (let i = startPage; i <= endPage; i++) addPageButton(i);

  if (endPage < totalPages) {

    if (endPage < totalPages - 1) addEllipsis();

    addPageButton(totalPages);

  }

  function addPageButton(pageNum) {

    const btn = document.createElement('button');

    btn.className = 'page-btn' + (pageNum === currentPage ? ' active' : '');

    btn.innerText = pageNum;

    btn.onclick = () => {

      currentPage = pageNum;

      update();

      document.querySelector('.cards')?.scrollIntoView({ behavior: 'smooth', block: 'start' });

    };

    paginationContainer.appendChild(btn);

  }

  function addEllipsis() {

    const span = document.createElement('span');

    span.innerText = '...';

    span.style.padding = '0 4px';

    span.style.color = '#94a3b8';

    paginationContainer.appendChild(span);

  }

}

function updateFiltersAndPagination() {

  let searchTerm = searchInput.value.toLowerCase().trim();

  let filtered = [...extendedTools];

  if (searchTerm !== "") filtered = filtered.filter(t => t.title.toLowerCase().includes(searchTerm) || t.desc.toLowerCase().includes(searchTerm));

  if (activeTag !== "all") filtered = filtered.filter(t => t.cat === activeTag);

  filteredTools = filtered;

  currentPage = 1;

  update();

}

function update() {

  renderTools();

  renderPagination();

}

perPageSelect.addEventListener('change', (e) => {

  perPage = parseInt(e.target.value, 10);

  currentPage = 1;

  update();

});

searchInput.addEventListener('input', () => updateFiltersAndPagination());

// TAG FILTERING

const tagElements = document.querySelectorAll('.tag');

tagElements.forEach(tag => {

  tag.addEventListener('click', () => {

    const tagValue = tag.getAttribute('data-tag');

    if (tagValue) {

      activeTag = tagValue;

      updateFiltersAndPagination();

      tagElements.forEach(t => { t.style.background = "#F1F5F9"; t.style.color = "#1E293B"; });

      tag.style.background = "#E0E7FF";

      tag.style.color = "#7C3AED";

    }

  });

});

// TYPING ANIMATION

const keywords = ["Keyword Research Tool", "SEO Audit Tool", "Meta Tags Generator", "Content Idea Generator", "Backlink Checker", "Topic Gap Finder"];

let kwIndex = 0, charPos = 0, isDeleting = false;

function typeEffect() {

  const inputField = searchInput;

  if (!inputField) return;

  const currentKeyword = keywords[kwIndex];

  if (isDeleting) {

    inputField.placeholder = currentKeyword.substring(0, charPos - 1);

    charPos--;

    if (charPos === 0) {

      isDeleting = false;

      kwIndex = (kwIndex + 1) % keywords.length;

      setTimeout(typeEffect, 200);

      return;

    }

    setTimeout(typeEffect, 45);

  } else {

    inputField.placeholder = currentKeyword.substring(0, charPos + 1);

    charPos++;

    if (charPos === currentKeyword.length) {

      isDeleting = true;

      setTimeout(typeEffect, 1800);

      return;

    }

    setTimeout(typeEffect, 85);

  }

}

function init() {

  update();

  typeEffect();

  const allTagElem = document.querySelector('.tag[data-tag="all"]');

  if(allTagElem) { allTagElem.style.background = "#E0E7FF"; allTagElem.style.color = "#7C3AED"; }

  activeTag = "all";

  filteredTools = [...extendedTools];

  update();

}

init();
